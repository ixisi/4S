-- The listener and TLS credentials stay intact when changing shared network.
!fn | chess_wait_next_network | {
    !if | net_role != 1 or net_lan_index == 0 or net_lan.count < 2 { !return |+| } |+|
    !var | net_lan_index | (net_lan_index % net_lan.count)+1 |+|
    !var | net_ip | net_lan.candidates[net_lan_index].address |+|
    !var | net_server | net_ip .. ":" .. net_port |+|
    !connection | invite | name:chess,address:net_ip,into:net_invite |+|
    !var | net_notice | "NETWORK CHANGED / COPY INVITE AGAIN" |+|
    !chess_wait_draw |+|
} |+|

!fn | chess_wait_input | {
    !var | next_wait_stage | (net_role == 1 and (net_peer > 0 and "PREPARING MATCH" or "WAITING FOR OPPONENT") or "CONNECTING TO HOST") |+|
    !if | net_role == 2 {
        !connection | status | name:chess,into:net_connection |+|
        !var | next_wait_stage | (net_connection.phase == "tcp" and "CONTACTING HOST" or
            net_connection.phase == "tls" and "SECURING CONNECTION" or
            net_connection.phase == "websocket" and "OPENING GAME CHANNEL" or "CHECKING INVITATION") |+|
    } |+|
    !if | net_role == 1 and net_peer == 0 {
        !connection | status | name:chess,into:net_connection |+|
        !if | net_connection.connections > 0 {
            !var | next_wait_stage | (net_connection.peers[1].phase == "invitation" and "CHECKING INVITATION" or
                net_connection.peers[1].phase == "websocket" and "OPENING GAME CHANNEL" or "SECURING CONNECTION") |+|
        } |+|
    } |+|
    !if | next_wait_stage != net_wait_stage { !var | net_wait_stage | next_wait_stage |+| !chess_wait_draw |+| } |+|
    !if | keys.TAB_pressed == 1 or keys.ESCAPE_pressed == 1 {
        !chess_net_lost | "ONLINE CONNECTION CANCELLED" |+| !return |+|
    } |+|
    !if | keys.LEFT_pressed == 1 or keys.RIGHT_pressed == 1 or keys.UP_pressed == 1 or keys.DOWN_pressed == 1 {
        !var | max_choice | (net_role == 1 and net_lan_index > 0 and net_lan.count > 1 and 3 or 2) |+|
        !var | net_wait_choice | net_wait_choice + ((keys.RIGHT_pressed == 1 or keys.DOWN_pressed == 1) and 1 or -1) |+|
        !if | net_wait_choice > max_choice { !var | net_wait_choice | (net_role == 1 and 1 or 2) |+| } |+|
        !if | net_wait_choice < (net_role == 1 and 1 or 2) { !var | net_wait_choice | max_choice |+| } |+|
        !chess_wait_draw |+|
    } |+|
    !if | keys.mouse_inside == 1 and keys.MOUSE_LEFT_pressed == 1 {
        !if | keys.x >= wait_cancel.x and keys.x < wait_cancel.x+wait_cancel.width and keys.y >= wait_cancel.y and keys.y < wait_cancel.y+wait_cancel.height {
            !chess_net_lost | "ONLINE CONNECTION CANCELLED" |+| !return |+|
        } |+|
        !if | net_role == 1 and keys.x >= wait_copy.x and keys.x < wait_copy.x+wait_copy.width and keys.y >= wait_copy.y and keys.y < wait_copy.y+wait_copy.height { !chess_net_copy_invite |+| } |+|
        !if | keys.x >= wait_network.x and keys.x < wait_network.x+wait_network.width and keys.y >= wait_network.y and keys.y < wait_network.y+wait_network.height { !chess_wait_next_network |+| } |+|
    } |+|
    !if | net_role == 1 and keys.C_pressed == 1 { !chess_net_copy_invite |+| } |+|
    !if | keys.N_pressed == 1 { !chess_wait_next_network |+| } |+|
    !if | keys.ENTER_pressed == 1 {
        !if | net_wait_choice == 1 and net_role == 1 { !chess_net_copy_invite |+| } |+|
        !if | net_wait_choice == 2 { !chess_net_lost | "ONLINE CONNECTION CANCELLED" |+| } |+|
        !if | net_wait_choice == 3 { !chess_wait_next_network |+| } |+|
    } |+|
} |+|
